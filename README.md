git init - инициализация репозитория
git status	Показывает изменённые файлы
git add .	Подготавливает все изменения к коммиту
git commit -m "..."	Сохраняет версию
git log --oneline	Показывает историю коммитов
git diff	Показывает незакоммиченные изменения
git push	Отправляет изменения на GitHub
git pull	Получает изменения
git clone URL	Скачивает репозиторий
git switch -c feature	Создаёт новую ветку и переключается на неё
git remote add origin https://github.com/USERNAME/DroneStudio.git - Подключение к удаленному репозиторию
git branch -M main - переименовывает текущую ветку в main
